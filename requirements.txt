# numerically-sensitive core (the suite currently runs on numpy 2.x / scipy 1.x).

# auto-detects PyQt6; older matplotlib only knows Qt5Agg.

# 0.13 is the first pyqtgraph release with PyQt6 support.
pyqtgraph>=0.13.0
# 6.7 is the floor the code targets. The old `<6.11` cap for Python 3.9 is gone
# with the 3.9 leg: PyQt6 6.11 ships cp310-abi3 wheels, and the baseline is now
# 3.10 (pyproject requires-python, CI matrix, ruff target-version = py310).
PyQt6>=6.7
scipy>=1.6.0,<2.0
Pillow>=8.0.0  # GIF animation export
